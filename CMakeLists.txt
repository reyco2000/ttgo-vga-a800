cmake_minimum_required(VERSION 3.16)
project(a800 C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(A800_UNDOCUMENTED_OPCODES "Enable stable undocumented NMOS 6502 opcodes" ON)
option(A800_BASIC "Include Atari BASIC ROM support" ON)
option(A800_NTSC "Include NTSC video support" ON)
option(A800_PAL "Include PAL video support" ON)
option(A800_SAVE_STATE "Include diagnostic save-state support" OFF)
option(A800_OPEN_EDITOR "Open the OS screen editor before direct XEX loading" ON)
if(NOT A800_NTSC AND NOT A800_PAL)
  message(FATAL_ERROR "At least one of A800_NTSC or A800_PAL must be enabled")
endif()
set(A800_CORE 6502.c Atari.c Antic.c AtariIo.c Gtia.c Pia.c Pokey.c)
if(A800_SAVE_STATE)
  list(APPEND A800_CORE State.c)
endif()
add_library(a800_core STATIC ${A800_CORE})
target_include_directories(a800_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT A800_BASIC)
  target_compile_definitions(a800_core PUBLIC NO_BASIC=1)
endif()
if(NOT A800_NTSC)
  target_compile_definitions(a800_core PUBLIC NO_NTSC=1)
endif()
if(NOT A800_PAL)
  target_compile_definitions(a800_core PUBLIC NO_PAL=1)
endif()
if(NOT A800_SAVE_STATE)
  target_compile_definitions(a800_core PUBLIC NO_SAVE_STATE=1)
endif()
if(NOT A800_OPEN_EDITOR)
  target_compile_definitions(a800_core PUBLIC NO_OPEN_EDITOR=1)
endif()
if(A800_UNDOCUMENTED_OPCODES)
  target_compile_definitions(a800_core PUBLIC A800_UNDOCUMENTED_OPCODES=1)
endif()
if(UNIX)
  target_link_libraries(a800_core PUBLIC m)
endif()
option(A800_BUILD_SDL "Build the SDL2 application" ON)
option(A800_BUILD_VALIDATION "Build the headless validation utility" ON)
if(A800_BUILD_SDL)
  find_package(SDL2 REQUIRED)
  add_executable(a800 A800.c)
  target_link_libraries(a800 PRIVATE a800_core)
  if(TARGET SDL2::SDL2)
    target_link_libraries(a800 PRIVATE SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(a800 PRIVATE SDL2::SDL2main)
    endif()
  else()
    target_include_directories(a800 PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(a800 PRIVATE ${SDL2_LIBRARIES})
  endif()
endif()
if(A800_BUILD_VALIDATION)
  add_executable(a800-validate tests/a800-validate.c)
  target_link_libraries(a800-validate PRIVATE a800_core)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE a800_core)
  add_test(NAME core_test COMMAND core_test)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME rom2xex_test COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/rom2xex_test.py)
  endif()
endif()
