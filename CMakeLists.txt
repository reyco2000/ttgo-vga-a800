cmake_minimum_required(VERSION 3.16)
project(a800 C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(A800_UNDOCUMENTED_OPCODES "Enable stable undocumented NMOS 6502 opcodes" ON)
set(A800_CORE 6502.c Atari.c Antic.c AtariIo.c Gtia.c Pia.c Pokey.c)
add_library(a800_core STATIC ${A800_CORE})
target_include_directories(a800_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(A800_UNDOCUMENTED_OPCODES)
  target_compile_definitions(a800_core PUBLIC A800_UNDOCUMENTED_OPCODES=1)
endif()
if(UNIX)
  target_link_libraries(a800_core PUBLIC m)
endif()
option(A800_BUILD_SDL "Build the SDL2 application" ON)
option(A800_BUILD_VALIDATION "Build the headless validation utility" ON)
if(A800_BUILD_SDL)
  find_package(SDL2 REQUIRED)
  add_executable(a800 A800.c)
  target_link_libraries(a800 PRIVATE a800_core)
  if(TARGET SDL2::SDL2)
    target_link_libraries(a800 PRIVATE SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(a800 PRIVATE SDL2::SDL2main)
    endif()
  else()
    target_include_directories(a800 PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(a800 PRIVATE ${SDL2_LIBRARIES})
  endif()
endif()
if(A800_BUILD_VALIDATION)
  add_executable(a800-validate tests/a800-validate.c)
  target_link_libraries(a800-validate PRIVATE a800_core)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE a800_core)
  add_test(NAME core_test COMMAND core_test)
endif()
